function doPost(e) {
  try {
    var data = JSON.parse(e.postData.contents);
    var ss = SpreadsheetApp.getActiveSpreadsheet();
    
    var action = data.action;

    // --- ПРОВЕРКА ИНН ---
    if (action === "check_inn") {
      var innToCheck = String(data.inn).trim();
      var ttSheet = ss.getSheetByName("Клиентская база");
      
      if (!ttSheet) {
        return createJsonResponse({ success: false, message: "Лист 'Клиентская база' не найден в базе данных!" });
      }

      var ttRows = ttSheet.getDataRange().getValues();
      
      // Ищем ИНН в колонке D (индекс 3)
      for (var k = 1; k < ttRows.length; k++) {
        var dbInn = String(ttRows[k][2]).trim(); 
        if (dbInn === innToCheck) {
          return createJsonResponse({
            success: true,
            exists: true,
            pointCode: ttRows[k][0],   // колонка A — Код
            pointName: ttRows[k][1]    // колонка B — Название
          });
        }
      }

      return createJsonResponse({ success: true, exists: false });
    }

    // --- ВХОД (АВТОРИЗАЦИЯ) ---
    if (action === "login") {
      var login = String(data.login || '').trim().toLowerCase();
      var password = String(data.password || '').trim();

      if (!login || !password) {
        return createJsonResponse({ success: false, message: "Заполните логин и пароль" });
      }

      // Лист с пользователями: "Пользователи", колонки A=Login, B=Password
      var usersSheetName = "Пользователи";
      var userSheet = ss.getSheetByName(usersSheetName);
      
      if (!userSheet) {
        return createJsonResponse({ success: false, message: "Лист с пользователями не найден!" });
      }

      var rows = userSheet.getDataRange().getValues();

      for (var j = 1; j < rows.length; j++) {
        var dbLogin = String(rows[j][0]).trim().toLowerCase();   // колонка A
        var dbPassword = String(rows[j][1]).trim();              // колонка B

        if (dbLogin === login && dbPassword === password) {
          return createJsonResponse({
            success: true,
            message: "Успешный вход!",
            user: { login: dbLogin }
          });
        }
      }
      return createJsonResponse({ success: false, message: "Неверный логин или пароль" });
    }

    // --- ПРИКРЕПЛЕНИЕ ТОЧКИ ---
    if (action === "attach") {
       var attachSheet = ss.getSheetByName("Прикрепление");
       var agentBrand = String(data.agent || '').substring(0, 2).toUpperCase();

       attachSheet.appendRow([
         data.pointCode,
         data.pointName,
         agentBrand,
         data.agent,
         data.visitDay,
         new Date()
       ]);

       return createJsonResponse({ success: true, message: "Точка успешно прикреплена к вам!" });
    }
    // --- ДОБАВЛЕНИЕ НОВОЙ ТОЧКИ ---
    if (action === "add_tt") {
       var addSheet = ss.getSheetByName("Добавление");

       addSheet.appendRow([
         data.clientName,
         data.geo,          // НОВОЕ: Геометка (широта, долгота)
         data.address,
         data.phone,
         data.inn,
         data.region,
         data.oblast,
         data.okrug,
         data.rayon,
         data.format,
         data.channel,
         data.type,
         data.category,
         data.deliveryCode,
         data.agent,
         data.visitDay,
         data.comments
       ]);

       return createJsonResponse({ success: true, message: "Новая торговая точка успешно добавлена!" });
    }

    return createJsonResponse({ success: false, message: "Неизвестное действие" });

  } catch(err) {
    return createJsonResponse({ success: false, message: err.toString() });
  }
}

function createJsonResponse(data) {
  return ContentService.createTextOutput(JSON.stringify(data))
    .setMimeType(ContentService.MimeType.JSON);
}
